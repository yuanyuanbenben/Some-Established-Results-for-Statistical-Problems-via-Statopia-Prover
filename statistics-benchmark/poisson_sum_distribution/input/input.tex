\begin{theorem}[Sum of Independent Poisson Variables]
Let $n\in\mathbb{N}$ with $n\ge 1$. For $i=1,\ldots,n$, let
$\lambda_i\in\mathbb{R}$ satisfy
\[
\lambda_i>0.
\]
Let $X_1,\ldots,X_n$ be independent $\mathbb{N}$-valued random variables
such that, for every $m\in\mathbb{N}$,
\[
P(X_i=m)=e^{-\lambda_i}\frac{\lambda_i^m}{m!}.
\]
Define
\[
S=\sum_{i=1}^n X_i,
\qquad
\Lambda=\sum_{i=1}^n\lambda_i.
\]
Then, for every $k\in\mathbb{N}$,
\[
P(S=k)=e^{-\Lambda}\frac{\Lambda^k}{k!}.
\]
\end{theorem}
