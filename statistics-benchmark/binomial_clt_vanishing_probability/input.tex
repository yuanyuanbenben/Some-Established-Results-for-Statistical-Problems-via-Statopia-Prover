\begin{theorem}[Binomial CLT with vanishing success probability]
Let $(\Omega,\mathcal A,P)$ be a probability space.  For every integer
$n\ge 1$, define
\[
  p_n=\frac{1}{\sqrt n},
\]
and let $Y_n:\Omega\to\mathbb N$ be an almost-everywhere measurable
random variable with the binomial distribution
\[
  Y_n\sim\operatorname{Binomial}(n,p_n).
\]
Equivalently, for $k=0,\ldots,n$,
\[
  P(Y_n=k)=\binom nk p_n^k(1-p_n)^{n-k},
\]
and $P(Y_n=k)=0$ for $k>n$.  No independence assumption between
different members of the sequence $(Y_n)_{n\ge 1}$ is required.

Define the scaled deviation
\[
  W_n=n^{3/4}\left(\frac{Y_n}{n}-p_n\right).
\]
Let $Z:\Omega\to\mathbb R$ be a measurable random variable with the
standard normal distribution $\mathcal N(0,1)$.  Then, as $n\to\infty$,
\[
  W_n\xrightarrow{\mathrm d}Z.
\]
\end{theorem}
