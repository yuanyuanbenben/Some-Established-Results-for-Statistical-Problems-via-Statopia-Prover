\begin{theorem}[Unbiasedness of the Empirical Distribution Function]
Let $n \in \mathbb{N}$ with $n \ge 1$. Let $X_1,\ldots,X_n$ be identically
distributed random variables with distribution function $F$. Let $t \in \mathbb{R}$.
Define
\[
F_n(t)=\frac{1}{n}\sum_{i=1}^n \mathbf{1}\{X_i \le t\}.
\]
Then
\[
\mathbb{E}[F_n(t)]=F(t).
\]
\end{theorem}
