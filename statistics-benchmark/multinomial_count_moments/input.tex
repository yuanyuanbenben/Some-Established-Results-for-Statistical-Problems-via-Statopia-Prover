\begin{theorem}[Multinomial Count Mean, Variance, and Covariance]
Let $k,n\in\mathbb{N}$ with $k\ge 1$. Let
$p_1,\ldots,p_k\in\mathbb{R}$ satisfy $p_a\ge 0$ for every $a$ and
\[
\sum_{a=1}^k p_a=1.
\]
Run $n$ independent categorical trials with category probabilities
$p_1,\ldots,p_k$. Let $N_a$ be the number of trials falling in category $a$.
Then, for every category $a$,
\[
\mathbb E[N_a]=np_a,
\qquad
\operatorname{Var}(N_a)=np_a(1-p_a),
\]
and for distinct categories $a\ne b$,
\[
\operatorname{Cov}(N_a,N_b)=-np_ap_b.
\]
\end{theorem}
