\begin{theorem}[Asymptotic Normality of the Sample Variance]
Let $(\Omega,\mathcal A,P)$ be a probability space and let
$(X_i)_{i\ge0}$ be a sequence of measurable, independent real-valued random
variables, each having the same distribution as $X_0$.  Let
$\mu,v,m_4\in\mathbb R$ and assume
\[
 \int |X_0|^4dP<\infty,\qquad
 \int X_0\,dP=\mu,
\]
\[
 \int(X_0-\mu)^2dP=v,\qquad
 \int(X_0-\mu)^4dP=m_4,\qquad
 v^2<m_4.
\]
Let $G:\Omega\to\mathbb R$ be measurable and have Gaussian distribution
$\mathcal N(0,m_4-v^2)$.  For $n\in\mathbb N$, define
\[
 \overline X_n=\frac1n\sum_{i=0}^{n-1}X_i,
 \qquad
 V_n=\frac1n\sum_{i=0}^{n-1}(X_i-\overline X_n)^2,
\]
using the convention $0^{-1}=0$.  Then, as $n\to\infty$,
\[
  \sqrt n\,(V_n-v)\xrightarrow{\mathrm d}G
\]
under $P$.
\end{theorem}
