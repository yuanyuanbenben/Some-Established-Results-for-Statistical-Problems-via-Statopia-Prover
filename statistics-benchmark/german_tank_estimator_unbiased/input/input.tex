\begin{theorem}[Unbiasedness of the German tank estimator]
Let $(\Omega,\mathcal F,P)$ be a probability space, and let $N,n\in\mathbb N$
satisfy $1\leq n\leq N$. Let $M:\Omega\to\mathbb N$ be measurable and
assume that $n\leq M(\omega)\leq N$ for every $\omega$. Suppose that, for
each integer $r$ with $n\leq r\leq N$,
\[
 P(M=r)=\frac{{r-1\choose n-1}}{{N\choose n}}.
\]
Then
\[
 \mathbb E_P\!\left[\frac{n+1}{n}M-1\right]=N.
\]
\end{theorem}
