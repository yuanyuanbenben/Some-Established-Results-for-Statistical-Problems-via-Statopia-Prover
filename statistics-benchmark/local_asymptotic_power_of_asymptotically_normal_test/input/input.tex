\begin{theorem}[Local asymptotic power of an asymptotically normal test]
Let $(\Omega,\mathcal F)$ be a measurable space. For each $n\in\mathbb N$,
let $P_n$ be a probability measure on this space and let
$\widehat\theta_n:\Omega\to\mathbb R$ be measurable. Fix
$\theta_0,h,c,\sigma\in\mathbb R$ with $\sigma>0$. Assume that, as
$n\to\infty$, the distribution under $P_{n+1}$ of
\[
 \frac{\sqrt{n+1}}{\sigma}
 \left(\widehat\theta_{n+1}-
 \left(\theta_0+\frac{h}{\sqrt{n+1}}\right)\right)
\]
converges weakly to $N(0,1)$. Then
\[
 P_{n+1}\!\left(
 \frac{\sqrt{n+1}\,(\widehat\theta_{n+1}-\theta_0)}{\sigma}>c
 \right)
 \longrightarrow
 1-\Phi\!\left(c-\frac{h}{\sigma}\right),
\]
where $\Phi$ is the cumulative distribution function of $N(0,1)$.
\end{theorem}
