\begin{theorem}[Uniform convergence of quantiles]
Let $\mathcal U$ be a nonempty index set and let $p\in(0,1)$.  For every
$u\in\mathcal U$, let $F_u$ be a real Stieltjes function, and for every
integer $r\geq0$ let $F_{r,u}$ be a real Stieltjes function.  Assume that,
for every $u\in\mathcal U$ and every $r\geq0$,
\[
  \lim_{t\to-\infty}F_u(t)=0,
  \quad \lim_{t\to\infty}F_u(t)=1,
  \quad \lim_{t\to-\infty}F_{r,u}(t)=0,
  \quad \lim_{t\to\infty}F_{r,u}(t)=1,
\]
and that
\[
  \lim_{r\to\infty}\sup_{u\in\mathcal U}\sup_{t\in\mathbb R}
    |F_{r,u}(t)-F_u(t)|=0.
\]
Define
\[
  q_u=\inf\{t\in\mathbb R:p\leq F_u(t)\},
  \qquad
  q_{r,u}=\inf\{t\in\mathbb R:p\leq F_{r,u}(t)\}.
\]
Assume also that for every $\delta>0$ there is an $\eta>0$ such that, for
every $u\in\mathcal U$,
\[
  F_u(q_u-\delta)\leq p-\eta
  \quad\text{and}\quad
  p+\eta\leq F_u(q_u+\delta).
\]
Then
\[
  \lim_{r\to\infty}\sup_{u\in\mathcal U}|q_{r,u}-q_u|=0.
\]
\end{theorem}
