\begin{theorem}[Optimality bound for Fisher's linear discriminant]
Let $d\geq1$, let $\mu_0,\mu_1\in\mathbb R^d$ with
$\mu_0\ne\mu_1$, and let $W$ be a symmetric positive-definite real
$d\times d$ matrix.  Then every nonzero $a\in\mathbb R^d$ satisfies
\[
  \frac{\bigl(a^{\mathsf T}(\mu_1-\mu_0)\bigr)^2}
       {a^{\mathsf T}Wa}
  \leq
  (\mu_1-\mu_0)^{\mathsf T}W^{-1}(\mu_1-\mu_0).
\]
\end{theorem}
